// The printer behind `fixtures.ts`: Move test code, kept as text, not part of the package.
//
// To regenerate: copy `move/` to a scratch directory, append the code below (without these
// comment lines) to `packages/ghosts/tests/ghosts.move`, which it extends (a module takes one
// `extend` per mode, and the code uses that file's `setup_*`, `deploy`, `end_turn` and ranked
// helpers), then run in `packages/ghosts`:
//
//     sui move test print_ -s 2>&1 | grep -o 'FIX [A-Za-z0-9]* [0-9a-f]*'
//
// Each line is `FIX <name> <hex>`: the name is the key in `fixtures.ts`. The ids the tests name
// (`id*`) are printed too. `gameScript0` to `gameScript10` are the game after each step of the
// scripted match `parity.test.ts` plays in the sandbox. The friendly games (`gameFriendly*`,
// `account*`) are printed by the two friendly tests. The ranked values (`ranked*`, `rules`,
// `event*`, `notice*`, `accountRanked*`) by the ranked ones; `unitL<rifle><armor><utility>` are
// `unit::from_loadout` for all 36 loadouts (`n` for no utility). The lobby values (`lobby*`,
// `accountLobby*` and the ids) come from the second part, which goes at the end of
// `packages/ghosts/tests/lobby.move` (it extends `ghosts::lobby`, so it can read a lobby's bytes)
// and runs with `sui move test print_lobbies -s`. Nothing is added to the repo.
//
// `State`, `Ending`, `Recruit` and `Record` did not change with ranked play: their fixtures are
// the earlier printer's (`print_records`, and `stateDeploying` to `recruitGrenadier`).

use ghosts::effect;
use grid::direction::{up, left};
use std::{bcs::to_bytes, debug};
use sui::hex;

fun fix(name: vector<u8>, bytes: vector<u8>) {
    let mut line = b"FIX ".to_string();
    line.append(name.to_string());
    line.append(b" ".to_string());
    line.append(hex::encode(bytes).to_string());
    debug::print(&line);
}

fun fix_id(name: vector<u8>, id: ID) { fix(name, to_bytes(&id)) }

/// A recruit with a Standard Rifle, no armor, carrying the utility of `kind`
/// (none for 255), in the `Account` of the sender.
fun hire_kind(acc: &mut Account, kind: u8, ctx: &mut TxContext): ID {
    let mut recruit = recruit_module::new("Jane Doe", ctx);
    recruit.add_weapon(items::rifle(1, ctx));
    if (kind != 255) recruit.add_utility(items::utility(kind, ctx));
    let id = object::id(&recruit);
    acc.add_recruit(recruit);
    id
}

/// The `Seat` last sent to the `Account` of `owner`, as bytes, and back where it was.
fun seat_bytes(f: &mut Fixture, owner: address): vector<u8> {
    let sender = f.ts.sender();
    f.ts.next_tx(sender);
    let seat = f.ts.take_from_address<Seat>(owner);
    let bytes = to_bytes(&seat);
    ts::return_to_address(owner, seat);
    bytes
}

/// A game for stakes on Demo 1, one grenadier a side: Alice waits, Bob finds her, Alice goes
/// first (the draw undone, see `put_first`).
#[test]
fun print_stakes_games() {
    let mut fixture = setup_grenadiers();
    let f = &mut fixture;
    fix_id(b"idAlice", object::id(&f.alice));
    fix_id(b"idBob", object::id(&f.bob));
    fix_id(b"idAliceRecruit", f.alice_recruit);
    fix_id(b"idBobRecruit", f.bob_recruit);
    fix_id(b"idStakes", object::id(&f.mm));
    fix(b"accountIdle", to_bytes(&f.alice));

    f.ts.next_tx(ALICE);
    find_game(&mut f.mm, &mut f.alice, &f.preset, &f.rng, &f.clock, f.ts.ctx());
    fix(b"accountSearching", to_bytes(&f.alice));
    f.ts.next_tx(BOB);
    find_game(&mut f.mm, &mut f.bob, &f.preset, &f.rng, &f.clock, f.ts.ctx());
    fix(b"accountPlaying", to_bytes(&f.bob));
    f.alice_first();
    let alice = object::id_address(&f.alice);
    fix(b"seat", f.seat_bytes(alice));

    f.ts.next_tx(ALICE);
    let mut game = f.ts.take_shared<Game>();
    fix_id(b"idGame", object::id(&game));
    fix(b"gameDeploying", to_bytes(&game));
    let seat = f.seat();
    accept_seat(&mut f.alice, seat);
    f.clock.set_for_testing(1000);
    f.deploy(&mut game);
    fix(b"gamePlayingT0", to_bytes(&game));
    f.end_turn(&mut game);
    move_unit(
        &mut game,
        &f.bob,
        &f.clock,
        vector[6, 5, up!(), up!(), left!(), left!(), left!(), left!(), up!()],
        f.ts.ctx(),
    );
    fix(b"gamePlayingT1Moved", to_bytes(&game));
    f.clock.set_for_testing(91_001);
    f.ts.next_tx(BOB);
    next_turn(&mut game, &f.bob, &f.clock, f.ts.ctx());
    f.clock.set_for_testing(181_002);
    next_turn(&mut game, &f.bob, &f.clock, f.ts.ctx());
    fix(b"gamePlayingMissed", to_bytes(&game));
    ts::return_shared(game);
    fixture.finish();
}

/// Four recruits a side on an empty 8x8 map. Alice: smoke (2, 1), frags (2, 4), (2, 5),
/// (2, 6). Bob: smoke (6, 1), flashbang (6, 3), plain (5, 5) and (7, 7). Positions are
/// popped from the back, so they are listed in reverse of the squad's order.
fun print_script(prefix: vector<u8>) {
    let positions = vector[
        vector[2, 6],
        vector[2, 5],
        vector[2, 4],
        vector[2, 1],
        vector[7, 7],
        vector[5, 5],
        vector[6, 3],
        vector[6, 1],
    ];
    let mut n = 0u8;
    let mut fixture = setup_on!(
        |id| map::new(id, 8),
        positions,
        |acc, ctx| {
            n = n + 1;
            if (n == 1) {
                hire_kind(acc, SMOKE, ctx);
                hire_kind(acc, FRAG, ctx);
                hire_kind(acc, FRAG, ctx);
                hire_kind(acc, FRAG, ctx)
            } else {
                hire_kind(acc, SMOKE, ctx);
                hire_kind(acc, FLASHBANG, ctx);
                hire_kind(acc, 255, ctx);
                hire_kind(acc, 255, ctx)
            }
        },
    );
    let f = &mut fixture;
    let mut game = f.start();
    f.clock.set_for_testing(1000);

    // Bob deploys first, as the sandbox's opponent does
    f.ts.next_tx(BOB);
    let squad = ts::ids_for_address<Recruit>(object::id_address(&f.bob));
    deploy_squad(&mut game, &mut f.bob, &f.clock, tickets(squad), f.ts.ctx());
    f.ts.next_tx(ALICE);
    let squad = ts::ids_for_address<Recruit>(object::id_address(&f.alice));
    deploy_squad(&mut game, &mut f.alice, &f.clock, tickets(squad), f.ts.ctx());
    fix(name_of(prefix, b"0"), to_bytes(&game));

    // Alice smokes (3, 2): her own smoker only
    f.throw_effect(&mut game, true, SMOKE, vector[2, 1], vector[3, 2]);
    fix(name_of(prefix, b"1"), to_bytes(&game));
    f.end_turn(&mut game);
    fix(name_of(prefix, b"2"), to_bytes(&game));

    // Bob smokes (4, 1), refreshing Alice's smoker, then flashes (5, 4): the plain recruit at
    // (5, 5) comes before his own flashbanger at (6, 3) in the unit index
    f.throw_effect(&mut game, false, SMOKE, vector[6, 1], vector[4, 1]);
    fix(name_of(prefix, b"3"), to_bytes(&game));
    f.throw_effect(&mut game, false, FLASHBANG, vector[6, 3], vector[5, 4]);
    fix(name_of(prefix, b"4"), to_bytes(&game));
    f.end_turn(&mut game);
    fix(name_of(prefix, b"5"), to_bytes(&game));

    // Alice throws three frags at the dazed recruit on (5, 5): it falls to the third
    perform_grenade(&mut game, &f.alice, &f.rng, &f.clock, 2, 4, 5, 5, f.ts.ctx());
    fix(name_of(prefix, b"6"), to_bytes(&game));
    perform_grenade(&mut game, &f.alice, &f.rng, &f.clock, 2, 5, 5, 5, f.ts.ctx());
    fix(name_of(prefix, b"7"), to_bytes(&game));
    perform_grenade(&mut game, &f.alice, &f.rng, &f.clock, 2, 6, 5, 5, f.ts.ctx());
    fix(name_of(prefix, b"8"), to_bytes(&game));

    // Alice ends: every effect left runs out; Bob ends too
    f.end_turn(&mut game);
    fix(name_of(prefix, b"9"), to_bytes(&game));
    f.end_turn(&mut game);
    fix(name_of(prefix, b"10"), to_bytes(&game));

    ts::return_shared(game);
    fixture.finish();
}

#[test]
fun print_effects_script() { print_script(b"gameScript") }

fun name_of(prefix: vector<u8>, suffix: vector<u8>): vector<u8> {
    let mut p = prefix;
    p.append(suffix);
    p
}

/// A ranked game of three on Black Rain, Contractors both: Alice waits with the commitment to
/// `alice_squad` and `SALT`, Bob finds her at 1 s with `bob_squad`, Alice reveals at 2 s, Bob
/// wipes her out, and the game is wrapped up at 5 s; then Bob, above the default, resets his
/// rating at 9 s. Who moves first is the scenario's draw.
#[test]
fun print_ranked_game() {
    let mut fixture = setup_black_rain();
    let f = &mut fixture;
    fix_id(b"idRankedAlice", object::id(&f.alice));
    fix_id(b"idRankedBob", object::id(&f.bob));
    fix_id(b"idRankedThree", object::id(&f.three));
    fix_id(b"idRankedFour", object::id(&f.four));
    fix_id(b"idRankedLevel", object::id(&f.preset));
    fix(b"rules", to_bytes(&f.rules));
    fix(b"rankedSalt", SALT);
    fix(b"rankedAliceSquad", alice_squad());
    fix(b"rankedBobSquad", bob_squad());
    fix(b"rankedCommitment", to_bytes(&commitment(alice_squad())));

    f.queue_ranked(alice_squad());
    fix(b"accountRankedSearching", to_bytes(&f.alice));
    fix(b"rankedQueueOne", to_bytes(&f.three));

    f.clock.set_for_testing(1_000);
    f.find_ranked_with(bob_squad());
    let started = event::events_by_type<GameStarted>();
    fix(b"eventRankedStarted", to_bytes(&started[0]));
    fix(b"accountRankedFinder", to_bytes(&f.bob));
    fix(b"accountRankedRevealWaiting", to_bytes(&f.alice));
    let alice = object::id_address(&f.alice);
    fix(b"seatRanked", f.seat_bytes(alice));

    f.ts.next_tx(ALICE);
    let mut game = f.ts.take_shared<Game>();
    fix_id(b"idRankedGame", object::id(&game));
    fix(b"gameRankedRevealWait", to_bytes(&game));

    f.clock.set_for_testing(2_000);
    f.reveal_with(&mut game, alice_squad());
    fix(b"gameRankedPlaying", to_bytes(&game));
    fix(b"accountRankedPlaying", to_bytes(&f.alice));

    let alice_side = side_of(&game, object::id(&f.alice));
    wipe_out(&mut game, alice_side);
    fix(b"gameRankedOver", to_bytes(&game));
    f.clock.set_for_testing(5_000);
    f.ts.next_tx(BOB);
    f.complete(game);
    let ended = event::events_by_type<GameEnded>();
    fix(b"eventRankedEnded", to_bytes(&ended[0]));
    fix(b"accountRankedAfterLoss", to_bytes(&f.alice));
    fix(b"accountRankedAfterWin", to_bytes(&f.bob));

    f.clock.set_for_testing(9_000);
    f.ts.next_tx(BOB);
    account::reset_rating(&mut f.bob, &f.rules, &f.clock, f.ts.ctx());
    fix(b"accountRankedReset", to_bytes(&f.bob));
    fixture.finish();
}

/// `unit::from_loadout` of every loadout: rifles 1-3, armors 1-3, frag, flashbang, smoke and none,
/// with the recruit ID `0x1` and the owner `0x2`.
#[test]
fun print_loadout_units() {
    let utilities = vector[0u8, 1, 2, 255];
    let mut w = 1u8;
    while (w <= 3) {
        let mut a = 1u8;
        while (a <= 3) {
            utilities.do_ref!(|u| {
                let unit = unit::from_loadout(@0x1.to_id(), @0x2.to_id(), w, a, *u);
                let mut name = b"unitL";
                name.push_back(48 + w);
                name.push_back(48 + a);
                name.push_back(if (*u == 255) 110 else 48 + *u);
                fix(name, to_bytes(&unit));
                unit.destroy();
            });
            a = a + 1;
        };
        w = w + 1;
    };
}
